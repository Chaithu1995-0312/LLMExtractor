## 🔸 Subgraph Route

```python
@app.route("/api/graph/subgraph")
def subgraph():
    node_id = request.args.get("node_id")
    session = SessionLocal()

    nodes = session.execute(
        text("SELECT * FROM nexus_graph.nodes WHERE id = :id"),
        {"id": node_id}
    ).mappings().all()

    edges = session.execute(
        text("""
            SELECT * FROM nexus_graph.edges
            WHERE source_id = :id OR target_id = :id
        """),
        {"id": node_id}
    ).mappings().all()

    session.close()

    return jsonify({
        "nodes": nodes,
        "edges": edges
    })
```
