```tsx
import React, { useEffect, useState } from "react";
import { io, Socket } from "socket.io-client";

interface AuditEvent {
  timestamp: string;
  event: string;
  component: string;
  agent: string;
  model_tier?: string;
  cost?: {
    usd?: number;
    tokens_in?: number;
    tokens_out?: number;
  };
  decision?: {
    action?: string;
    reason?: string;
  };
}

const socket: Socket = io("http://localhost:5001");

export default function AuditStreamPanel() {
  const [events, setEvents] = useState<AuditEvent[]>([]);
  const [filter, setFilter] = useState<string>("");

  useEffect(() => {
    socket.on("connected", (msg) => {
      console.log(msg);
    });

    socket.on("audit_event", (data: AuditEvent) => {
      setEvents((prev) => [data, ...prev.slice(0, 200)]);
    });

    return () => {
      socket.off("audit_event");
    };
  }, []);

  const filtered = events.filter((e) =>
    filter ? e.component === filter : true
  );

  return (
    <div style={{ padding: 16 }}>
      <h2>Live Audit Stream</h2>

      <select
        value={filter}
        onChange={(e) => setFilter(e.target.value)}
      >
        <option value="">All Components</option>
        <option value="compiler">Compiler</option>
        <option value="graph">Graph</option>
        <option value="cognition">Cognition</option>
      </select>

      <div style={{ marginTop: 12, maxHeight: 500, overflowY: "scroll" }}>
        {filtered.map((e, index) => (
          <div
            key={index}
            style={{
              padding: 8,
              borderBottom: "1px solid #ddd",
              background:
                e.event.includes("HALLUCINATION")
                  ? "#ffe5e5"
                  : "#f9f9f9",
            }}
          >
            <strong>{e.event}</strong>
            <div>{e.component} | {e.agent}</div>
            <div>{new Date(e.timestamp).toLocaleTimeString()}</div>

            {e.cost?.usd && (
              <div>💰 ${e.cost.usd.toFixed(6)}</div>
            )}
          </div>
        ))}
      </div>
    </div>
  );
}
```

### 🔥 Implementation

```python
from collections import defaultdict
from datetime import datetime
from typing import Dict, Any

class CostAggregator:
    def __init__(self):
        self.cost_by_minute = defaultdict(float)
        self.cost_by_model = defaultdict(float)
        self.cost_by_topic = defaultdict(float)

    def process_event(self, event: Dict[str, Any]):
        cost = event.get("cost")
        if not cost or not cost.get("usd"):
            return

        usd = cost["usd"]
        ts = datetime.fromisoformat(event["timestamp"])
        minute_key = ts.strftime("%Y-%m-%d %H:%M")

        self.cost_by_minute[minute_key] += usd
        self.cost_by_model[event.get("model_tier", "unknown")] += usd
        self.cost_by_topic[event.get("topic_id", "unknown")] += usd

    def snapshot(self):
        return {
            "by_minute": dict(self.cost_by_minute),
            "by_model": dict(self.cost_by_model),
            "by_topic": dict(self.cost_by_topic)
        }
```

### 🔥 Implementation

```python
from collections import deque
import statistics

class AnomalyDetector:
    def __init__(self, window_size=50, threshold=3.0):
        self.window = deque(maxlen=window_size)
        self.threshold = threshold

    def process(self, value: float):
        self.window.append(value)
        if len(self.window) < 10:
            return False

        mean = statistics.mean(self.window)
        stdev = statistics.stdev(self.window)

        if stdev == 0:
            return False

        z_score = abs(value - mean) / stdev
        return z_score > self.threshold
```

```python
from datetime import datetime, timedelta

class SafetyController:
    def __init__(self):
        self.safe_mode = False
        self.triggered_at = None
        self.cooldown = timedelta(minutes=15)

    def trigger(self, reason: str):
        self.safe_mode = True
        self.triggered_at = datetime.utcnow()
        print(f"[SAFE MODE ENABLED] Reason: {reason}")

    def check(self):
        if not self.safe_mode:
            return False

        if datetime.utcnow() - self.triggered_at > self.cooldown:
            self.safe_mode = False
            print("[SAFE MODE DISABLED]")
            return False

        return True
```

```python
class AgentOrchestrator:
    def __init__(self, event_bus):
        self.bus = event_bus

    def subscribe(self):
        pubsub = self.bus.client.pubsub()
        pubsub.subscribe("nexus_pulse")

        for message in pubsub.listen():
            if message["type"] != "message":
                continue

            payload = json.loads(message["data"])

            self.route(payload)

    def route(self, payload):
        if payload["type"] == "CONTRADICTION":
```
